/**
 * Generate cryptographically secure refresh token
 */
function hlb_generate_refresh_token(): string {
    return bin2hex(random_bytes(64));
}

/**
 * Generate JWT access token for user
 */
function hlb_generate_access_token(WP_User $user): string {
    $issued_at = time();
    $expire = $issued_at + (defined('JWT_AUTH_EXPIRE') ? JWT_AUTH_EXPIRE : 900);

    $payload = [
        'iss'  => get_bloginfo('url'),
        'iat'  => $issued_at,
        'nbf'  => $issued_at,
        'exp'  => $expire,
        'data' => [
            'user' => [
                'id'    => $user->ID,
                'email' => $user->user_email,
                'roles' => $user->roles
            ]
        ]
    ];

    return \Firebase\JWT\JWT::encode($payload, JWT_AUTH_SECRET_KEY, 'HS256');
}

/**
 * Decode and validate JWT token
 */
function hlb_decode_access_token(string $token): ?object {
    try {
        return \Firebase\JWT\JWT::decode($token, JWT_AUTH_SECRET_KEY, ['HS256']);
    } catch (Exception $e) {
        return null;
    }
}

/**
 * Get current user from JWT Authorization header
 */
function hlb_get_user_from_token(): ?WP_User {
    $auth_header = $_SERVER['HTTP_AUTHORIZATION'] ?? $_SERVER['REDIRECT_HTTP_AUTHORIZATION'] ?? '';

    if (!preg_match('/Bearer\s+(.*)$/i', $auth_header, $matches)) {
        return null;
    }

    $token = trim($matches[1]);
    $decoded = hlb_decode_access_token($token);

    if (!$decoded || !isset($decoded->data->user->id)) {
        return null;
    }

    $user = get_user_by('id', $decoded->data->user->id);
    return ($user && $user->exists()) ? $user : null;
}

/**
 * Permission callback: Require valid JWT token
 */
function hlb_require_auth(): bool {
    return hlb_get_user_from_token() !== null;
}

/**
 * Permission callback: Require specific role
 */
function hlb_require_role(string $role): bool {
    $user = hlb_get_user_from_token();
    return $user && in_array($role, $user->roles);
}

/**
 * Permission callback: Require email verification
 */
function hlb_require_email_verified(): bool {
    $user = hlb_get_user_from_token();
    return $user && get_user_meta($user->ID, 'email_verified', true);
}

/**
 * Send JSON response with proper headers
 */
function hlb_respond($data, int $status = 200): void {
    status_header($status);
    header('Content-Type: application/json; charset=' . get_option('blog_charset'));
    echo wp_json_encode($data, JSON_UNESCAPED_SLASHES | JSON_PRETTY_PRINT);
    exit;
}

/**
 * Format property data for API response
 */
function hlb_format_property(WP_Post $post): array {
    $gallery_ids = get_field('gallery_images', $post->ID) ?: [];
    $gallery = array_filter(array_map('wp_get_attachment_url', $gallery_ids));

    $amenities = get_field('amenities', $post->ID) ?: [];
    $amenities = is_array($amenities) ? $amenities : [];

    return [
        'id'                    => $post->ID,
        'title'                 => get_the_title($post),
        'description'           => wpautop($post->post_content),
        'host_id'               => $post->post_author,
        'host_name'             => get_the_author_meta('display_name', $post->post_author),
        'price_per_night'       => (float) get_field('price_per_night', $post->ID),
        'cleaning_fee'          => (float) get_field('cleaning_fee', $post->ID),
        'service_fee_percent'   => (float) get_field('service_fee_percent', $post->ID),
        'max_guests'            => (int) get_field('max_guests', $post->ID),
        'bedrooms'              => (int) get_field('bedrooms', $post->ID),
        'bathrooms'             => (float) get_field('bathrooms', $post->ID),
        'address'               => get_field('address', $post->ID),
        'coordinates'           => get_field('coordinates', $post->ID),
        'amenities'             => $amenities,
        'house_rules'           => get_field('house_rules', $post->ID),
        'cancellation_policy'   => get_field('cancellation_policy', $post->ID),
        'instant_booking'       => (bool) get_field('instant_booking', $post->ID),
        'status'                => get_field('property_status', $post->ID) ?: HLB_PROPERTY_STATUS_DRAFT,
        'rating'                => (float) get_field('average_rating', $post->ID),
        'total_reviews'         => (int) get_field('total_reviews', $post->ID),
        'featured_image'        => get_the_post_thumbnail_url($post->ID, 'large'),
        'gallery'               => $gallery,
        'created_at'            => get_the_date('c', $post),
        'updated_at'            => get_the_modified_date('c', $post)
    ];
}

/**
 * Format booking data for API response
 */
function hlb_format_booking(WP_Post $post): array {
    $property_id = (int) get_field('property_id', $post->ID);
    $check_in = get_field('check_in_date', $post->ID);
    $check_out = get_field('check_out_date', $post->ID);

    $nights = $check_in && $check_out
        ? max(1, (strtotime($check_out) - strtotime($check_in)) / 86400)
        : 0;

    return [
        'id'                => $post->ID,
        'property_id'       => $property_id,
        'property_title'    => $property_id ? get_the_title($property_id) : null,
        'traveler_id'       => (int) get_field('traveler_id', $post->ID),
        'host_id'           => $property_id ? get_post_field('post_author', $property_id) : null,
        'check_in'          => $check_in,
        'check_out'         => $check_out,
        'nights'            => $nights,
        'price_per_night'   => (float) get_field('price_per_night', $property_id),
        'cleaning_fee'      => (float) get_field('cleaning_fee', $property_id),
        'service_fee'       => (float) get_field('service_fee', $post->ID),
        'taxes'             => (float) get_field('taxes', $post->ID),
        'total_price'       => (float) get_field('total_price', $post->ID),
        'currency'          => 'USD',
        'booking_status'    => get_field('booking_status', $post->ID),
        'payment_status'    => get_field('payment_status', $post->ID),
        'cancellation_deadline' => get_field('cancellation_deadline', $post->ID),
        'created_at'        => get_the_date('c', $post),
        'updated_at'        => get_the_modified_date('c', $post)
    ];
}

/**
 * Format user profile for API response
 */
function hlb_format_user_profile(WP_User $user): array {
    $avatar_id = get_user_meta($user->ID, 'profile_image_id', true);

    return [
        'id'                        => $user->ID,
        'email'                     => $user->user_email,
        'name'                      => $user->display_name,
        'first_name'                => $user->first_name,
        'last_name'                 => $user->last_name,
        'bio'                       => get_user_meta($user->ID, 'bio', true),
        'phone'                     => get_user_meta($user->ID, 'phone', true),
        'avatar_url'                => $avatar_id ? wp_get_attachment_url($avatar_id) : get_avatar_url($user->ID, ['size' => 256]),
        'roles'                     => $user->roles,
        'email_verified'            => (bool) get_user_meta($user->ID, 'email_verified', true),
        'host_verification_status'  => get_user_meta($user->ID, 'host_verification_status', true) ?: 'unverified',
        'identity_verification_status' => get_user_meta($user->ID, 'identity_verification_status', true) ?: 'unverified',
        'account_status'            => get_user_meta($user->ID, 'account_status', true) ?: 'active',
        'average_rating'            => (float) get_user_meta($user->ID, 'host_average_rating', true),
        'total_reviews'             => (int) get_user_meta($user->ID, 'host_total_reviews', true),
        'created_at'                => date('c', strtotime($user->user_registered)),
        'last_login'                => get_user_meta($user->ID, 'last_login', true)
    ];
}

/**
 * Format notification for API response
 */
function hlb_format_notification(WP_Post $post): array {
    return [

        'type'          => get_field('notification_type', $post->ID),

    ];
}

/**
 * Send verification email
 */
function hlb_send_verification_email(WP_User $user): bool {
    $key = wp_generate_password(20, false);
    update_user_meta($user->ID, 'email_verification_key', $key);
    update_user_meta($user->ID, 'email_verification_expiry', time() + 24 * HOUR_IN_SECONDS);

    $verify_url = rest_url('api/v1/auth/verify-email') . '?user_id=' . $user->ID . '&key=' . $key;

    return wp_mail(
        $user->user_email,
        'Verify your email - ' . get_bloginfo('name'),
        "Click to verify: $verify_url\n\nThis link expires in 24 hours."
    );
}

/**
 * Register cron jobs for background tasks
 */
function hlb_register_cron_jobs() {
    $events = [
        'hlb_expire_unpaid_bookings' => ['schedule' => 'hourly', 'action' => 'hlb_cron_expire_unpaid_bookings'],
        'hlb_mark_completed_bookings' => ['schedule' => 'daily', 'action' => 'hlb_cron_mark_completed_bookings'],
        'hlb_send_booking_reminders' => ['schedule' => 'daily', 'action' => 'hlb_cron_send_booking_reminders'],
        'hlb_clean_orphaned_data' => ['schedule' => 'weekly', 'action' => 'hlb_cron_clean_orphaned_data']
    ];

    foreach ($events as $hook => $config) {
        if (!wp_next_scheduled($config['action'])) {
            wp_schedule_event(time(), $config['schedule'], $config['action']);
        }
    }
}
add_action('init', 'hlb_register_cron_jobs');

/* =============================================================================
   REST API ROUTES REGISTRATION
============================================================================= */

add_action('rest_api_init', function () {

    /* -------------------------------------------------------------------------
       1️⃣ AUTHENTICATION DOMAIN
    -------------------------------------------------------------------------- */

    // POST /auth/register
    register_rest_route(HLB_API_NAMESPACE, '/auth/register', [
        'methods' => 'POST',
        'permission_callback' => '__return_true',
        'args' => [
            'name' => ['required' => true, 'type' => 'string', 'sanitize_callback' => 'sanitize_text_field'],
            'email' => ['required' => true, 'type' => 'string', 'validate_callback' => 'is_email'],
            'password' => ['required' => true, 'type' => 'string', 'min_length' => 8],
            'role' => ['required' => true, 'type' => 'string', 'enum' => ['traveler', 'host']]
        ],
        'callback' => function (WP_REST_Request $req) {
            $name = sanitize_text_field($req['name']);
            $email = sanitize_email($req['email']);
            $password = $req['password'];
            $role = sanitize_text_field($req['role']);

            if (email_exists($email)) {
                return new WP_Error('email_exists', 'Email already registered', ['status' => 409]);
            }

            if (strlen($password) < 8) {
                return new WP_Error('weak_password', 'Password must be at least 8 characters', ['status' => 400]);
            }

            $user_id = wp_create_user($email, $password, $email);
            if (is_wp_error($user_id)) {
                return new WP_Error('registration_failed', 'Could not create user', ['status' => 500]);
            }

            $user = new WP_User($user_id);
            $user->set_role($role);
            wp_update_user(['ID' => $user_id, 'display_name' => $name, 'first_name' => explode(' ', $name)[0]]);

            // Send verification email
            hlb_send_verification_email($user);

            // Log registration
            update_user_meta($user_id, 'registration_ip', $_SERVER['REMOTE_ADDR'] ?? '');
            update_user_meta($user_id, 'registration_user_agent', $_SERVER['HTTP_USER_AGENT'] ?? '');

            return [
                'success' => true,
                'message' => 'Registration successful. Please check your email to verify your account.',
                'user_id' => $user_id,
                'email_verified' => false
            ];
        }
    ]);

    // POST /auth/verify-email
    register_rest_route(HLB_API_NAMESPACE, '/auth/verify-email', [
        'methods' => 'GET',
        'permission_callback' => '__return_true',
        'args' => [
            'user_id' => ['required' => true, 'type' => 'integer'],
            'key' => ['required' => true, 'type' => 'string']
        ],
        'callback' => function (WP_REST_Request $req) {
            $user_id = (int) $req['user_id'];
            $key = sanitize_text_field($req['key']);
            $user = get_user_by('id', $user_id);

            if (!$user) {
                return new WP_Error('user_not_found', 'User not found', ['status' => 404]);
            }

            $stored_key = get_user_meta($user_id, 'email_verification_key', true);
            $expiry = get_user_meta($user_id, 'email_verification_expiry', true);

            if (!$stored_key || $key !== $stored_key || time() > $expiry) {
                return new WP_Error('invalid_key', 'Invalid or expired verification link', ['status' => 400]);
            }

            update_user_meta($user_id, 'email_verified', true);
            delete_user_meta($user_id, 'email_verification_key');
            delete_user_meta($user_id, 'email_verification_expiry');

            // Create welcome notification
            hlb_create_notification($user_id, 'welcome', 'Welcome!', 'Your account has been verified. Start exploring!');

            return [
                'success' => true,
                'message' => 'Email verified successfully',
                'email_verified' => true
            ];
        }
    ]);

    // POST /auth/login
    register_rest_route(HLB_API_NAMESPACE, '/auth/login', [
        'methods' => 'POST',
        'permission_callback' => '__return_true',
        'args' => [
            'email' => ['required' => true, 'type' => 'string'],
            'password' => ['required' => true, 'type' => 'string']
        ],
        'callback' => function (WP_REST_Request $req) {
            $email = sanitize_email($req['email']);
            $password = $req['password'];

            $user = wp_authenticate($email, $password);
            if (is_wp_error($user)) {
                sleep(1); // Rate limiting
                return new WP_Error('invalid_credentials', 'Invalid email or password', ['status' => 401]);
            }

            // Check account status
            $account_status = get_user_meta($user->ID, 'account_status', true) ?: 'active';
            if ($account_status !== 'active') {
                return new WP_Error('account_inactive', 'Account is ' . $account_status, ['status' => 403]);
            }

            // Update last login
            update_user_meta($user->ID, 'last_login', current_time('mysql'));

            // Generate tokens
            $access_token = hlb_generate_access_token($user);
            $refresh_token = hlb_generate_refresh_token();

            // Store refresh token hash
            update_user_meta($user->ID, 'refresh_token_hash', wp_hash_password($refresh_token));
            update_user_meta($user->ID, 'refresh_token_expiry', time() + JWT_REFRESH_EXPIRE);
            update_user_meta($user->ID, 'refresh_token_user_agent', $_SERVER['HTTP_USER_AGENT'] ?? '');

            return [
                'access_token' => $access_token,
                'refresh_token' => $refresh_token,
                'token_type' => 'Bearer',
                'expires_in' => JWT_AUTH_EXPIRE,
                'user' => hlb_format_user_profile($user)
            ];
        }
    ]);

    // POST /auth/refresh
    register_rest_route(HLB_API_NAMESPACE, '/auth/refresh', [
        'methods' => 'POST',
        'permission_callback' => '__return_true',
        'args' => [
            'refresh_token' => ['required' => true, 'type' => 'string'],
            'user_id' => ['required' => true, 'type' => 'integer']
        ],
        'callback' => function (WP_REST_Request $req) {
            $refresh_token = sanitize_text_field($req['refresh_token']);
            $user_id = (int) $req['user_id'];

            $hash = get_user_meta($user_id, 'refresh_token_hash', true);
            $expiry = get_user_meta($user_id, 'refresh_token_expiry', true);

            if (!$hash || time() > $expiry || !wp_check_password($refresh_token, $hash)) {
                // Token rotation: invalidate old token on misuse
                if ($hash) {
                    delete_user_meta($user_id, 'refresh_token_hash');
                    delete_user_meta($user_id, 'refresh_token_expiry');
                }
                return new WP_Error('invalid_refresh', 'Invalid or expired refresh token', ['status' => 401]);
            }

            $user = get_user_by('id', $user_id);
            if (!$user || $user->ID !== $user_id) {
                return new WP_Error('user_not_found', 'User not found', ['status' => 404]);
            }

            // Rotate refresh token
            $new_refresh = hlb_generate_refresh_token();
            update_user_meta($user_id, 'refresh_token_hash', wp_hash_password($new_refresh));
            update_user_meta($user_id, 'refresh_token_expiry', time() + JWT_REFRESH_EXPIRE);

            return [
                'access_token' => hlb_generate_access_token($user),
                'refresh_token' => $new_refresh,
                'expires_in' => JWT_AUTH_EXPIRE
            ];
        }
    ]);

    // POST /auth/logout
    register_rest_route(HLB_API_NAMESPACE, '/auth/logout', [
        'methods' => 'POST',
        'permission_callback' => 'hlb_require_auth',
        'callback' => function () {
            $user = hlb_get_user_from_token();
            if (!$user) {
                return new WP_Error('unauthorized', 'Not authenticated', ['status' => 401]);
            }

            // Invalidate refresh token
            delete_user_meta($user->ID, 'refresh_token_hash');
            delete_user_meta($user->ID, 'refresh_token_expiry');

            // Could add token blacklist here for immediate access token invalidation

            return ['success' => true, 'message' => 'Logged out successfully'];
        }
    ]);

    // GET /auth/me
    register_rest_route(HLB_API_NAMESPACE, '/auth/me', [
        'methods' => 'GET',
        'permission_callback' => 'hlb_require_auth',
        'callback' => function () {
            $user = hlb_get_user_from_token();
            if (!$user) {
                return new WP_Error('unauthorized', 'Not authenticated', ['status' => 401]);
            }
            return hlb_format_user_profile($user);
        }
    ]);

    // POST /auth/forgot-password
    register_rest_route(HLB_API_NAMESPACE, '/auth/forgot-password', [
        'methods' => 'POST',
        'permission_callback' => '__return_true',
        'args' => ['email' => ['required' => true, 'type' => 'string']],
        'callback' => function (WP_REST_Request $req) {
            $email = sanitize_email($req['email']);
            $user = get_user_by('email', $email);

            // Always return success to prevent user enumeration
            if (!$user) {
                return ['success' => true, 'message' => 'If that email exists, a reset link has been sent'];
            }

            // Use WordPress core password reset
            retrieve_password($user->user_login);

            return ['success' => true, 'message' => 'If that email exists, a reset link has been sent'];
        }
    ]);

    // POST /auth/reset-password
    register_rest_route(HLB_API_NAMESPACE, '/auth/reset-password', [
        'methods' => 'POST',
        'permission_callback' => '__return_true',
        'args' => [
            'key' => ['required' => true, 'type' => 'string'],
            'login' => ['required' => true, 'type' => 'string'],
            'new_password' => ['required' => true, 'type' => 'string', 'min_length' => 8]
        ],
        'callback' => function (WP_REST_Request $req) {
            $key = sanitize_text_field($req['key']);
            $login = sanitize_user($req['login']);
            $new_password = $req['new_password'];

            if (strlen($new_password) < 8) {
                return new WP_Error('weak_password', 'Password must be at least 8 characters', ['status' => 400]);
            }

            $user = check_password_reset_key($key, $login);
            if (is_wp_error($user)) {
                return new WP_Error('invalid_key', 'Invalid or expired reset key', ['status' => 400]);
            }

            reset_password($user, $new_password);

            // Invalidate all refresh tokens for security
            delete_user_meta($user->ID, 'refresh_token_hash');
            delete_user_meta($user->ID, 'refresh_token_expiry');

            return ['success' => true, 'message' => 'Password reset successfully'];
        }
    ]);

    // POST /auth/upgrade-role
    register_rest_route(HLB_API_NAMESPACE, '/auth/upgrade-role', [
        'methods' => 'POST',
        'permission_callback' => 'hlb_require_auth',
        'args' => [
            'target_role' => ['required' => true, 'type' => 'string', 'enum' => ['host']]
        ],
        'callback' => function (WP_REST_Request $req) {
            $user = hlb_get_user_from_token();
            if (!$user) {
                return new WP_Error('unauthorized', 'Not authenticated', ['status' => 401]);
            }

            if (!in_array('traveler', $user->roles)) {
                return new WP_Error('invalid_state', 'Only travelers can upgrade to host', ['status' => 400]);
            }

            // Start host verification process
            update_user_meta($user->ID, 'host_verification_status', 'pending');
            $user->add_role('host');

            // Notify admin
            hlb_create_notification(1, 'admin_alert', 'New Host Application', "User {$user->ID} requested host upgrade");

            return [
                'success' => true,
                'message' => 'Host application submitted. Verification pending.',
                'host_verification_status' => 'pending'
            ];
        }
    ]);

    /* -------------------------------------------------------------------------
       2️⃣ USER / PROFILE DOMAIN

                'paged' => (int) $req['page'],

                $args['s'] = sanitize_text_field($req['search']);

            // Price filter

                $args['meta_query'][] = [
                    'key' => 'price_per_night',
                    'value' => [(float) ($req['min_price'] ?: 0), (float) ($req['max_price'] ?: PHP_FLOAT_MAX)],
                    'type' => 'NUMERIC',
                    'compare' => 'BETWEEN'
                ];
            }

            // Guests filter
            if ($req['min_guests']) {
                $args['meta_query'][] = [
                    'key' => 'max_guests',
                    'value' => (int) $req['min_guests'],
                    'type' => 'NUMERIC',
                    'compare' => '>='
                ];

            // Amenities filter
            if ($req['amenities'] && is_array($req['amenities'])) {
                $amenities = array_map('sanitize_text_field', $req['amenities']);
                foreach ($amenities as $amenity) {
                    $args['meta_query'][] = [
                        'key' => 'amenities',
                        'value' => '"' . $amenity . '"',
                        'compare' => 'LIKE'
                    ];
                }

            if ($req['instant_booking'] !== null) {
                $args['meta_query'][] = [
                    'key' => 'instant_booking',
                    'value' => $req['instant_booking'] ? '1' : '0'

                    $args['order'] = 'DESC';

                    $args['order'] = 'DESC';

    // GET /properties/{id}/availability
    register_rest_route(HLB_API_NAMESPACE, '/properties/(?P<id>\d+)/availability', [
        'methods' => 'GET',
        'permission_callback' => '__return_true',
        'args' => [
            'id' => ['required' => true, 'type' => 'integer'],
            'check_in' => ['type' => 'string', 'format' => 'date'],
            'check_out' => ['type' => 'string', 'format' => 'date'],
            'guests' => ['type' => 'integer']
        ],
        'callback' => function (WP_REST_Request $req) {
            $property_id = (int) $req['id'];
            $post = get_post($property_id);
            if (!$post) {
                return new WP_Error('not_found', 'Property not found', ['status' => 404]);
            }

            $min_stay = (int) get_field('minimum_stay', $property_id) ?: 1;
            $max_stay = (int) get_field('maximum_stay', $property_id) ?: 30;
            $advance_notice = (int) get_field('advance_booking_days', $property_id) ?: 0;
            $blocked_dates = get_field('blocked_dates', $property_id) ?: [];

            // If date range requested, check availability
            if ($req['check_in'] && $req['check_out']) {
                $check_in = strtotime(sanitize_text_field($req['check_in']));
                $check_out = strtotime(sanitize_text_field($req['check_out']));
                $nights = ($check_out - $check_in) / 86400;

                if ($check_out <= $check_in) {
                    return new WP_Error('invalid_dates', 'Check-out must be after check-in', ['status' => 400]);
                }
                if ($nights < $min_stay) {
                    return new WP_Error('min_stay', "Minimum stay is {$min_stay} nights", ['status' => 400]);
                }
                if ($nights > $max_stay) {
                    return new WP_Error('max_stay', "Maximum stay is {$max_stay} nights", ['status' => 400]);
                }
                if ($check_in < strtotime("+{$advance_notice} days")) {
                    return new WP_Error('advance_notice', "Booking requires {$advance_notice} days advance notice", ['status' => 400]);
                }

                // Check blocked dates
                for ($t = $check_in; $t < $check_out; $t += 86400) {
                    $date = date('Y-m-d', $t);
                    if (in_array($date, $blocked_dates)) {
                        return new WP_Error('date_blocked', "Date {$date} is not available", ['status' => 409]);
                    }
                }

                // Check existing bookings (confirmed only)

                    'post_type' => HLB_BOOKING_POST_TYPE,
                    'meta_query' => [
                        ['key' => 'property_id', 'value' => $property_id],
                        ['key' => 'booking_status', 'value' => HLB_BOOKING_STATUS_CONFIRMED]
                    ]
                ]);

                foreach ($existing->posts as $booking) {
                    $b_in = strtotime(get_field('check_in_date', $booking->ID));
                    $b_out = strtotime(get_field('check_out_date', $booking->ID));

                    // Overlap detection
                    if ($check_in < $b_out && $check_out > $b_in) {
                        return new WP_Error('date_conflict', 'Dates conflict with existing booking', ['status' => 409]);
                    }
                }

                // Calculate price
                $base_price = (float) get_field('price_per_night', $property_id);
                $cleaning_fee = (float) get_field('cleaning_fee', $property_id);
                $service_percent = (float) get_field('service_fee_percent', $property_id) ?: 14;

                $subtotal = $base_price * $nights;
                $service_fee = $subtotal * ($service_percent / 100);
                $taxes = $subtotal * 0.1; // Example tax rate
                $total = $subtotal + $cleaning_fee + $service_fee + $taxes;

                return [
                    'available' => true,
                    'property_id' => $property_id,
                    'check_in' => date('Y-m-d', $check_in),
                    'check_out' => date('Y-m-d', $check_out),
                    'nights' => $nights,
                    'pricing' => [
                        'base_price' => $base_price,
                        'nights' => $nights,
                        'subtotal' => $subtotal,
                        'cleaning_fee' => $cleaning_fee,
                        'service_fee' => $service_fee,
                        'taxes' => $taxes,
                        'total' => $total,
                        'currency' => 'USD'
                    ],
                    'rules' => [
                        'min_stay' => $min_stay,
                        'max_stay' => $max_stay,
                        'advance_notice' => $advance_notice,
                        'instant_booking' => (bool) get_field('instant_booking', $property_id),
                        'cancellation_policy' => get_field('cancellation_policy', $property_id)
                    ]
                ];
            }

            // Return calendar view
            return [
                'property_id' => $property_id,
                'blocked_dates' => $blocked_dates,
                'min_stay' => $min_stay,
                'max_stay' => $max_stay,
                'booking_window' => [
                    'advance_days' => $advance_notice,
                    'max_advance_days' => (int) get_field('max_advance_booking_days', $property_id) ?: 365

            if ($user->ID !== $traveler_id) {

            if (!in_array($status, [HLB_BOOKING_STATUS_PENDING_PAYMENT, HLB_BOOKING_STATUS_PENDING_APPROVAL, HLB_BOOKING_STATUS_CONFIRMED])) {

            $deadline = get_field('cancellation_deadline', $booking_id);

            if ($status === HLB_BOOKING_STATUS_CONFIRMED && time() > strtotime($deadline)) {
                // Late cancellation: charge fee
                $total = (float) get_field('total_price', $booking_id);
                $cancellation_fee = $total * 0.5; // 50% penalty example

                'booking_status' => $new_status,

    // POST /reviews
    register_rest_route(HLB_API_NAMESPACE, '/reviews', [
        'methods' => 'POST',
        'permission_callback' => function () {
            return hlb_require_auth() && hlb_require_role('traveler');
        },
        'args' => [
            'booking_id' => ['required' => true, 'type' => 'integer'],
            'rating' => ['required' => true, 'type' => 'integer', 'minimum' => 1, 'maximum' => 5],
            'comment' => ['type' => 'string', 'sanitize_callback' => 'sanitize_textarea_field']
        ],
        'callback' => function (WP_REST_Request $req) {
            $user = hlb_get_user_from_token();
            $booking_id = (int) $req['booking_id'];
            $booking = get_post($booking_id);

            if (!$booking || $booking->post_type !== HLB_BOOKING_POST_TYPE) {
                return new WP_Error('not_found', 'Booking not found', ['status' => 404]);
            }

            $traveler_id = (int) get_field('traveler_id', $booking->ID);
            if ($user->ID !== $traveler_id) {
                return new WP_Error('forbidden', 'Only the traveler can review this booking', ['status' => 403]);
            }

            // Check booking status and completion
            $booking_status = get_field('booking_status', $booking_id);
            $check_out = get_field('check_out_date', $booking_id);

            if ($booking_status !== HLB_BOOKING_STATUS_COMPLETED) {
                return new WP_Error('not_allowed', 'Can only review completed bookings', ['status' => 403]);
            }

            if (strtotime($check_out) > time()) {
                return new WP_Error('too_early', 'Cannot review before checkout date', ['status' => 403]);
            }

            // Prevent duplicate reviews
            if (get_field('reviewed', $booking_id)) {
                return new WP_Error('duplicate', 'Review already submitted for this booking', ['status' => 409]);
            }

            $property_id = (int) get_field('property_id', $booking_id);
            $rating = (int) $req['rating'];
            $comment = sanitize_textarea_field($req['comment']);

            // Create review
            $review_id = wp_insert_post([
                'post_type' => HLB_REVIEW_POST_TYPE,
                'post_status' => 'publish',
                'post_title' => 'Review for Property #' . $property_id,
                'post_content' => $comment,
                'post_author' => $user->ID
            ]);

            update_field('rating', $rating, $review_id);
            update_field('property_id', $property_id, $review_id);
            update_field('booking_id', $booking_id, $review_id);
            update_field('reviewed', true, $booking_id);

            // Update property average rating
            $reviews = new WP_Query([
                'post_type' => HLB_REVIEW_POST_TYPE,
                'meta_query' => [['key' => 'property_id', 'value' => $property_id]],
                'fields' => 'ids'
            ]);

            $ratings = array_map(fn($id) => (int) get_field('rating', $id), $reviews->posts);
            $average = $ratings ? array_sum($ratings) / count($ratings) : 0;

            update_field('average_rating', round($average, 2), $property_id);
            update_field('total_reviews', count($ratings), $property_id);

            // Update host average rating
            $host_id = get_post_field('post_author', $property_id);
            $host_reviews = new WP_Query([
                'post_type' => HLB_REVIEW_POST_TYPE,
                'meta_query' => [
                    ['key' => 'property_id', 'value' => $property_id],
                    ['key' => 'post_author', 'compare' => 'EXISTS']
                ]
            ]);
            // Simplified: in production, query all properties by host
            $host_avg = get_user_meta($host_id, 'host_average_rating', true) ?: 0;
            $host_total = (int) get_user_meta($host_id, 'host_total_reviews', true);
            $new_avg = (($host_avg * $host_total) + $rating) / ($host_total + 1);

            update_user_meta($host_id, 'host_average_rating', round($new_avg, 2));
            update_user_meta($host_id, 'host_total_reviews', $host_total + 1);

            // Notify host
            hlb_create_notification($host_id, 'new_review', 'New Review Received', "You received a {$rating}-star review for booking #{$booking_id}");

            return [
                'success' => true,
                'review_id' => $review_id,
                'property_rating_updated' => round($average, 2)
            ];
        }
    ]);

// Rate limiting helper (basic implementation)
function hlb_rate_limit(string $key, int $limit, int $window): bool {
    $transient = 'hlb_rate_' . md5($key);
    $attempts = get_transient($transient) ?: 0;

    if ($attempts >= $limit) {
        return false;
    }

    set_transient($transient, $attempts + 1, $window);
    return true;
}

// Add rate limiting to auth endpoints
add_filter('rest_pre_dispatch', function ($result, $server, $request) {
    $path = $request->get_route();

    // Rate limit auth endpoints: 5 attempts per 15 minutes per IP
    if (strpos($path, '/auth/login') !== false || strpos($path, '/auth/register') !== false) {
        $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
        if (!hlb_rate_limit("auth:{$ip}", 5, 15 * MINUTE_IN_SECONDS)) {
            return new WP_Error('rate_limited', 'Too many attempts. Please try again later.', ['status' => 429]);
        }
    }

    return $result;
}, 10, 3);

// CORS headers for headless frontend
add_filter('rest_pre_serve_request', function ($served, $result, $request) {
    header('Access-Control-Allow-Origin: *'); // Restrict in production
    header('Access-Control-Allow-Methods: GET, POST, PUT, DELETE, OPTIONS');
    header('Access-Control-Allow-Headers: Authorization, Content-Type, Idempotency-Key');

    // Handle preflight
    if ($_SERVER['REQUEST_METHOD'] === 'OPTIONS') {
        status_header(200);
        exit;
    }

    return $served;
}, 10, 3);

add_action('rest_api_init', function () {
    if (defined('WP_DEBUG') && WP_DEBUG) {
        add_filter('rest_json_encode_options', function ($options) {
            $options['depth'] = 20;
            return $options;
        });
    }
});

    wp_clear_scheduled_hook('hlb_cron_send_booking_reminders');

    flush_rewrite_rules();
});
